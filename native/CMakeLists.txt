cmake_minimum_required(VERSION 3.20)
project(phasechange_native LANGUAGES CXX)
include(CTest)
add_library(phasechange_native SHARED kernels.cpp)
target_compile_features(phasechange_native PUBLIC cxx_std_17)
target_compile_definitions(phasechange_native PRIVATE PC_BUILD)
target_include_directories(phasechange_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(phasechange_native PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(phasechange_native PRIVATE /W4 /WX)
else()
  target_compile_options(phasechange_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/library-$<CONFIG>.txt"
     CONTENT "$<TARGET_FILE:phasechange_native>\n")
if(BUILD_TESTING)
  add_executable(phasechange_invariants test_kernels.cpp)
  target_link_libraries(phasechange_invariants PRIVATE phasechange_native)
  add_test(NAME phasechange_native_invariants COMMAND phasechange_invariants)
endif()
